\begin{frame}{Operational Setting and Research Questions}
    \small
    {\color{structure.fg}\textbf{Operational setting}}
    \vspace{0.2em}
    \begin{itemize}
        \setlength\itemsep{0.6em}
        \item Configuring \(\theta\) can be costly; in practice, it is often fixed once.
        \item But the network context changes over time: traffic volume, composition, topology, and attack types evolve.
    \end{itemize}

    \vspace{0.55em}
    {\color{structure.fg}\textbf{Research questions}}
    \vspace{0.2em}
    \begin{itemize}
        \setlength\itemsep{0.6em}
        \item Can cheaper strategies approach an oracle-optimized reference without full reconfiguration?
        \item If \(\theta\) is reused, how much performance is lost compared with reconfiguring for the current network context?
    \end{itemize}

    \vfill
    \centering
    {\color{structure.fg}\small
        \(\Longrightarrow\) Measure the performance penalty of avoiding full reconfiguration in AD IDS under changing network contexts.}
\end{frame}

        \centering
        {\color{structure.fg}\small
            \only<1>{\(\Longrightarrow\) Why nested? Choose \(\theta\) using only training folds, avoiding test leakage.}
            \only<2>{\(\Longrightarrow\) Configure \(\theta\): BO validates on inner benign folds + one attack training fold; reuse skips BO.}
            \only<3>{\(\Longrightarrow\) Evaluation: IF trains on benign folds, tests on held-out folds, and gives one \(p_{\text{robust}}\).}
            \only<4>{\(\Longrightarrow\) Repeat the outer loop over the five folds before averaging.}
            \only<5->{\(\Longrightarrow\) The reported score is mean \(p_{\text{robust}}\).}}
    \end{minipage}
