\section{Conclusion}

\begin{frame}{Takeaway}
    \small
    \begin{columns}[T,onlytextwidth]
        \column{0.57\linewidth}
        \begin{itemize}
            \setlength\itemsep{0.8em}
            \item Retraining alone does not guarantee robustness when hyperparameters stay fixed.
            \item Configuration reuse under context changes can remove roughly one third of discrimination power.
            \item AD IDS need automated reconfiguration, not a one-time setup.
        \end{itemize}

        \column{0.38\linewidth}
        \setbeamercolor{takebox}{use={structure,normal text},
            fg=normal text.fg,
            bg=structure.fg!10!normal text.bg}
        \begin{beamercolorbox}[wd=\linewidth,sep=1.2ex,rounded=true]{takebox}
            \textbf{Next direction}\par
            \vspace{0.35em}
            Trigger reconfiguration only when a lightweight change detector reports enough traffic change.
        \end{beamercolorbox}
    \end{columns}

    \vfill
    \centering
    {\color{structure.fg}\normalsize
        \(\Longrightarrow\) Treat the configuration as an operational asset to monitor.}
\end{frame}
